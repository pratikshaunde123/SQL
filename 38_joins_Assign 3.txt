	from emp a,emp b

q.6)write a query for the inner join of the dept and emp table, display DEPTNO,DNAME, name
    select d.deptno, d.dname, e.ename
     from dept d ,emp e
    where d.deptno = e.deptno;

q.7)display on unique deptno from emp and dname.

q.8)display the information from the emp table for all the employees for which the deptno is same as
that of smith.

select *
from emp
where deptno = (select deptno
                from emp
               where ename = 'SMITH');

q.9) display the information from the emp table for all the employees whose sal is greater than the
average sal of the same deptno.

select *
from emp e
where sal > (select AVG(sal)
             from emp
             where deptno = e.deptno);

q.10)display the dept number and the avg salary department wise.

select deptno,avg(sal) AS average_salary
from emp
group by deptno;
